def preorder_traversal(root):
    # 定义先序遍历函数，参数 root 为当前子树的根节点（或 None）
    if root is None:
        # 如果根节点为空，说明到达了空子树，直接返回空列表
        return []
    else:
        # 否则，当前节点非空，需要处理该节点及其左右子树
        result = []  # 初始化结果列表，用于存储当前子树遍历得到的节点值
        result.append(root.data)  # 先访问根节点，将根节点的值加入结果列表（先序：根-左-右）
        result.extend(preorder_traversal(root.left))   # 递归遍历左子树，并将返回的列表元素扩展到 result 中
        result.extend(preorder_traversal(root.right))  # 递归遍历右子树，并将返回的列表元素扩展到 result 中
        return result  # 返回当前子树完整的先序遍历序列
