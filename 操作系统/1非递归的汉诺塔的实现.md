def hanoi(from_: int, to_: int, num_plate: int):
    if num_plate == 1:
        print(f"{from_} -> {to_}.")
    else:
        via_ = 3 - from_ - to_          # 计算中间柱（编号 0,1,2 之和为 3）
        hanoi(from_, via_, num_plate - 1)   # 将上方 n-1 个盘子移到辅助柱
        hanoi(from_, to_, 1)                # 将最大盘子移到目标柱
        hanoi(via_, to_, num_plate - 1)     # 将辅助柱上的 n-1 个盘子移到目标柱

其中的`via_` 是辅助柱，`from_` 是起始柱，`to_` 是目标柱，`num_plate` 是盘子的数量。

import matplotlib
matplotlib.use('TkAgg')  # 根据您的系统可改为 'Qt5Agg'
import matplotlib.pyplot as plt
from matplotlib.patches import Rectangle
import copy

class HanoiVisualizer:
    def __init__(self):
        plt.ion()
        self.fig, self.ax = plt.subplots(figsize=(6, 4))
        self._setup_base_and_pillars()
        self.step_text = self.ax.text(0.5, 4.5, 'Step: 0', fontsize=12, ha='center')
        plt.show(block=False)
        self.fig.canvas.draw()

    def _setup_base_and_pillars(self):
        self.ax.set_xlim(-3, 11)
        self.ax.set_ylim(-1, 5)
        self.ax.axis('off')
        # 底座
        self.ax.add_patch(Rectangle((-3, -0.5), 14, 0.6, facecolor='#404040', edgecolor='black', lw=2, zorder=0))
        # 柱子
        for x in [0, 4, 8]:
            self.ax.add_patch(Rectangle((x-0.25, -0.5+0.6), 0.5, 4, facecolor='#8B4513', edgecolor='black', lw=1, zorder=1))

    def update(self, state, step):
        # 移除旧圆盘（仅移除 zorder=2 的矩形）
        for patch in list(self.ax.patches):
            if isinstance(patch, Rectangle) and patch.get_zorder() == 2:
                patch.remove()

        columns_x = [0, 4, 8]
        all_disks = [d for tower in state for d in tower]
        max_disk = max(all_disks) if all_disks else 1
        scale = 3.0 / max_disk
        disk_height = 0.4
        cmap = plt.cm.plasma

        for col_idx, tower in enumerate(state):
            x_center = columns_x[col_idx]
            total = len(tower)
            # 从底部开始绘制，所以直接按列表顺序（底部→顶部）
            for level, disk in enumerate(tower):
                y_base = -0.5 + 0.6 + (total - level - 1) * disk_height
                width = disk * scale
                color = cmap((disk - 1) / (max_disk - 1)) if max_disk > 1 else cmap(0)
                self.ax.add_patch(Rectangle(
                    (x_center - width/2, y_base), width, disk_height,
                    facecolor=color, edgecolor='black', lw=1, zorder=2
                ))

        self.step_text.set_text(f'Step: {step}')
        self.fig.canvas.draw_idle()
        self.fig.canvas.flush_events()


# ========== 状态生成器（递归） ==========
def generate_hanoi_states(n, source=0, target=2, auxiliary=1):
    """
    生成汉诺塔的状态序列（列表形式）。
    返回一个列表，每个元素为三个列表的元组，表示三根柱子上的盘子（底部→顶部）。
    """
    # 初始状态：所有盘子在 source 柱，按从大到小（底部→顶部）
    initial_state = [
        list(range(n, 0, -1)),  # 源柱：大盘在下，小盘在上
        [],
        []
    ]
    states = [copy.deepcopy(initial_state)]

    def _move(src, dst, aux, count):
        if count == 1:
            # 执行移动：从 src 柱顶部取出一个盘子，放到 dst 柱顶部
            disk = states[-1][src].pop()
            states[-1][dst].append(disk)
            states.append(copy.deepcopy(states[-1]))  # 记录移动后的状态
        else:
            _move(src, aux, dst, count - 1)
            _move(src, dst, aux, 1)
            _move(aux, dst, src, count - 1)

    _move(source, target, auxiliary, n)
    return states


# ========== 主程序 ==========
if __name__ == "__main__":
    n = 5  # 盘子数量
    states = generate_hanoi_states(n, source=0, target=2, auxiliary=1)

    visualizer = HanoiVisualizer()

    # 显示初始状态（step=0）
    visualizer.update(states[0], 0)
    plt.pause(1.0)

    # 依次显示每一步（从第一步开始）
    for step in range(1, len(states)):
        visualizer.update(states[step], step)
        plt.pause(0.8)  # 每步间隔 0.8 秒

    plt.ioff()
    plt.show()  # 保持窗口打开
